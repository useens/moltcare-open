class SensenMCPClient:

#### 任务调度算法

```python
class TaskScheduler:
    """智能任务调度器 - 根据任务类型和节点状态分配"""
    
    NODE_CAPABILITIES = {
        'cloud': {
            'strengths': ['thinking', 'coordination', 'memory_write'],
            'resources': {'cpu': 4, 'memory': 23, 'gpu': False},
            'latency': 'low'  # 到外部服务
        },
        'local': {
            'strengths': ['execution', 'computation', 'processing'],
            'resources': {'cpu': 8, 'memory': 16, 'gpu': True},
            'latency': 'ultra_low'  # 本地执行
        }
    }
    
    def schedule(self, task: Task) -> NodeSelection:
        """根据任务特性选择最佳节点"""
        
        # 1. 检查任务类型偏好
        if task.type in ['thinking', 'decision', 'coordination']:
            return self.NODE_CAPABILITIES['cloud']
        
        if task.type in ['compilation', 'training', 'batch_processing']:
            return self.NODE_CAPABILITIES['local']
        
        # 2. 检查资源需求
        if task.requires_gpu or task.cpu_intensive:
            return self.NODE_CAPABILITIES['local']
        
        # 3. 检查网络依赖
        if task.external_apis:
            return self.NODE_CAPABILITIES['cloud']
        
        # 4. 负载均衡
        return self.select_by_load()
```

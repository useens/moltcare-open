## 📋 项目概述

**Moltcare** 是一个双AI自动协作框架，旨在让所有 OpenClaw agents 一键提升智能水平。

### 核心使命

1. **双AI协作**: 两个AI自动分工、协作、互相审查
2. **智能增强**: 新安装的 agents 自动获得智能提升
3. **专家讨论**: 重要开发阶段强制多专家参与

### 讨论策略实现

```typescript
// src/discussion/strategies/round-robin.ts

export class RoundRobinStrategy implements IDiscussionStrategy {
  private currentIndex = 0;
  
  async executeRound(
    experts: ExpertAgent[],
    context: DiscussionContext
  ): Promise<RoundResult> {
    const currentExpert = experts[this.currentIndex];
    
    // 专家发言
    const contribution = await currentExpert.contribute(context);
    
    // 更新索引
    this.currentIndex = (this.currentIndex + 1) % experts.length;
    
    return {
      speaker: currentExpert,
      contribution,
      nextSpeaker: experts[this.currentIndex]
    };
  }
}

// src/discussion/strategies/debate.ts

export class DebateStrategy implements IDiscussionStrategy {
  async executeRound(
    experts: ExpertAgent[],
    context: DiscussionContext
  ): Promise<RoundResult> {
    // 分组辩论
    const [proSide, conSide] = this.splitExperts(experts);
    
    // 正方发言
    const proArgs = await Promise.all(
      proSide.map(e => e.argue(context, Position.PRO))
    );
    
    // 反方反驳
    const conArgs = await Promise.all(
      conSide.map(e => e.argue(context, Position.CON))
    );
    
    // 评估论点强度
    return this.evaluateArguments(proArgs, conArgs);
  }
}
```

### 共识构建机制

```typescript
// src/discussion/consensus-builder.ts

export class ConsensusBuilder {
  // 共识阈值
  private readonly CONSENSUS_THRESHOLD = 0.7;
  
  async buildConsensus(
    opinions: ExpertOpinion[],
    strategy: ConsensusStrategy
  ): Promise<ConsensusResult> {
    switch (strategy) {
      case 'voting':
        return this.votingConsensus(opinions);
      case 'weighted':
        return this.weightedConsensus(opinions);
      case 'delphi':
        return this.delphiMethod(opinions);
      default:
        throw new Error(`Unknown consensus strategy: ${strategy}`);
    }
  }
  
  private async votingConsensus(opinions: ExpertOpinion[]): Promise<ConsensusResult> {
    // 简单多数投票
    const votes = this.countVotes(opinions);
    const winner = this.findMajority(votes);
    
    const agreement = winner.count / opinions.length;
    
    return {
      decision: winner.option,
      confidence: agreement,
      isConsensus: agreement >= this.CONSENSUS_THRESHOLD,
      dissentingOpinions: this.getDissentingOpinions(opinions, winner.option)
    };
  }
  
  private async delphiMethod(opinions: ExpertOpinion[]): Promise<ConsensusResult> {
    // 德尔菲法：多轮匿名反馈直到收敛
    let round = 1;
    let currentOpinions = opinions;
    
    while (round <= this.MAX_ROUNDS) {
      // 匿名汇总
      const summary = this.anonymizeAndSummarize(currentOpinions);
      
      // 专家重新评估
      const newOpinions = await Promise.all(
        currentOpinions.map(o => this.reEvaluate(o, summary))
      );
      
      // 检查收敛
      if (this.hasConverged(newOpinions)) {
        return this.formConsensus(newOpinions);
      }
      
      currentOpinions = newOpinions;
      round++;
    }
    
    // 达到最大轮数，强制结论
    return this.forceConclusion(currentOpinions);
  }
}
```

  triggers: ['complex-query', 'high-volume']
